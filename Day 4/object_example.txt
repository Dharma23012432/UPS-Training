class Student {

    String name;
    int age;
    long mobile;

    Student(String name, int age,long mobile) {
        this.name = name;
        this.age = age;
        this.mobile = mobile;
    }

    void display(Student s) {
        System.out.println("Name : " + s.name);
        System.out.println("Age  : " + s.age);
        System.out.println("Mobile No : " + s.mobile);

        System.out.println();
    }
}

public class Main {

    public static void main(String[] args) {

        Student s1 = new Student("Tony", 64,7598746958L);
        Student s2 = new Student("Steve", 45,5846932184L);
        Student s3 = new Student("Banner", 42,9547868547L);

        s1.display(s1);
        s2.display(s2);
        s3.display(s3);
    }
}
